function parseVehiclePrice(price) {

    if (typeof price === "number") {
        return price;
    }

    return Number(
        String(price)
            .replace(/Rp/g, "")
            .replace(/\./g, "")
            .replace(/,/g, "")
            .trim()
    ) || 0;

}


function formatRupiah(number) {

    return "Rp " +
        Number(number || 0).toLocaleString(
            "id-ID"
        );

}

/* =====================================================
   SUBMIT SIMULATION
===================================================== */

function submitSimulation() {

    const vehicleKey =
        localStorage.getItem("tafSimulationVehicle");

    if (!vehicleKey) {

        alert("Unit kendaraan tidak ditemukan.");
        return;

    }


    const vehicle =
        vehicleData[vehicleKey];

    if (!vehicle) {

        alert("Data kendaraan tidak ditemukan.");
        return;

    }


    const dpInput =
        document.getElementById("dp");

    const tenorInput =
        document.getElementById("tenor");

    const insuranceInput =
        document.getElementById("insurance");

    const branchInput =
        document.getElementById("branch");


    const dp =
        Number(
            dpInput ? dpInput.value : 20
        );


    const tenor =
        Number(
            tenorInput ? tenorInput.value : 60
        );


    const insurance =
        insuranceInput
            ? insuranceInput.value
            : "";


    const branch =
        branchInput
            ? branchInput.value
            : "";


    /* =========================================
       VALIDATION
    ========================================= */

    if (dp < 15) {

        alert(
            "Uang muka minimum 15%."
        );

        return;

    }


    if (!insurance) {

        alert(
            "Silakan pilih jenis asuransi."
        );

        return;

    }


    if (!branch) {

        alert(
            "Silakan pilih Branch TAF."
        );

        return;

    }


    /* =========================================
       BASIC CALCULATION
    ========================================= */

    const vehiclePrice =
        parseVehiclePrice(
            vehicle.price
        );


    const dpAmount =
        vehiclePrice *
        (dp / 100);


    const financing =
        vehiclePrice -
        dpAmount;


    /* =========================================
       INSURANCE
    ========================================= */

    const insuranceResult =
        calculateInsurancePremium(
            vehiclePrice,
            tenor,
            insurance
        );


    const insurancePremium =
        insuranceResult.premium;


    /* =========================================
       PH
    ========================================= */

    const ph =
        financing +
        insurancePremium;


    /* =========================================
       PROTOTYPE INTEREST
    ========================================= */

    /*
     * Prototype estimation only.
     * Bukan bunga resmi TAF.
     */

    const estimatedRate =
        0.08;


    const estimatedInterest =
        ph *
        estimatedRate *
        (tenor / 12);


    /* =========================================
       TOTAL AR
    ========================================= */

    const estimatedTotal =
        ph +
        estimatedInterest;


    /* =========================================
       INSTALLMENT
    ========================================= */

    const estimatedInstallment =
        estimatedTotal /
        tenor;


    /* =========================================
       TDP
    ========================================= */

    /*
     * Prototype:
     * Untuk sementara TDP = DP.
     *
     * Biaya lain belum dimasukkan karena
     * belum ada sumber data resmi pada prototype.
     */

    const tdp =
        dpAmount;


    /* =========================================
       SIMULATION DATA
    ========================================= */

    const simulationData = {

        vehicle: vehicleKey,

        vehicleName: vehicle.name,

        vehiclePrice: vehiclePrice,

        brand: vehicle.brand,

        model: vehicle.model,

        type: vehicle.type,

        year: vehicle.year,

        mileage: vehicle.mileage,

        transmission: vehicle.transmission,

        fuel: vehicle.fuel,

        color: vehicle.color,

        showroom: vehicle.showroom,

        dpPercentage: dp,

        dpAmount: dpAmount,

        tenor: tenor,

        insurance: insurance,

        insuranceProvider:
            "Asuransi Astra",

        insuranceCoverage:
            insuranceResult.coverage,

        insuranceRate:
            insuranceResult.rate,

        insurancePremium:
            insurancePremium,

        branch: branch,

        financing: financing,

        ph: ph,

        tdp: tdp,

        estimatedRate:
            estimatedRate,

        estimatedInterest:
            estimatedInterest,

        estimatedTotal:
            estimatedTotal,

        estimatedInstallment:
            estimatedInstallment,

        createdAt:
            new Date().toISOString()

    };


    /* =========================================
       SAVE SIMULATION
    ========================================= */

    localStorage.setItem(
        "tafSimulation",
        JSON.stringify(
            simulationData
        )
    );


    /* =========================================
       GO TO RESULT
    ========================================= */

    window.location.href =
        "hasil-simulasi.html";

}

/* =====================================================
   CATEGORY FILTER
===================================================== */

function filterCategory(category, button) {

    const cards =
        document.querySelectorAll(".vehicle-card");

    const buttons =
        document.querySelectorAll(".category");

    const emptyState =
        document.getElementById("emptyState");

    let visibleCount = 0;


    /* ---------------------------------------------
       RESET ACTIVE CATEGORY
    --------------------------------------------- */

    buttons.forEach(function(item) {

        item.classList.remove("active");

    });


    if (button) {

        button.classList.add("active");

    }


    /* ---------------------------------------------
       FILTER VEHICLE
    --------------------------------------------- */

    cards.forEach(function(card) {

        const cardCategory =
            card.dataset.category;


        if (
            category === "Semua" ||
            cardCategory === category
        ) {

            card.style.display = "";

            visibleCount++;

        } else {

            card.style.display = "none";

        }

    });


    /* ---------------------------------------------
       EMPTY STATE
    --------------------------------------------- */

    if (!emptyState) {
        return;
    }


    if (visibleCount === 0) {

        emptyState.style.display =
            "block";

    } else {

        emptyState.style.display =
            "none";

    }

}


/* =====================================================
   SEARCH VEHICLE
===================================================== */

function searchVehicle() {

    const input =
        document.getElementById("searchInput");

    const cards =
        document.querySelectorAll(".vehicle-card");

    const emptyState =
        document.getElementById("emptyState");


    if (!input) {
        return;
    }


    const keyword =
        input.value
            .toLowerCase()
            .trim();


    let visibleCount = 0;


    cards.forEach(function(card) {

        const name =
            card.dataset.name
                ? card.dataset.name.toLowerCase()
                : "";


        if (
            name.includes(keyword) ||
            keyword === ""
        ) {

            card.style.display = "";

            visibleCount++;

        } else {

            card.style.display = "none";

        }

    });


    if (!emptyState) {
        return;
    }


    if (visibleCount === 0) {

        emptyState.style.display =
            "block";

    } else {

        emptyState.style.display =
            "none";

    }

}

/* =====================================================
   INSURANCE PROTOTYPE
===================================================== */

/*
 * CATATAN:
 * Rate di bawah adalah rate PROTOTYPE,
 * bukan rate resmi TAF / Asuransi Astra.
 *
 * Rate resmi dapat diganti kemudian
 * apabila sudah tersedia sumber resmi.
 */

const simulationInsuranceRate = {

    comprehensive: {

        12: 0.0255,
        24: 0.0297,
        36: 0.0297,
        48: 0.0297,
        60: 0.0297

    },

    tlo: {

        12: 0.0084,
        24: 0.0102,
        36: 0.0102,
        48: 0.0102,
        60: 0.0102

    }

};


const simulationDepreciation = {

    12: 1.00,
    24: 0.80,
    36: 0.80,
    48: 0.70,
    60: 0.70

};


/* =====================================================
   GET INSURANCE RATE
===================================================== */

function getInsuranceRate(
    insuranceType,
    tenor
) {

    if (
        simulationInsuranceRate[
            insuranceType
        ] &&
        simulationInsuranceRate[
            insuranceType
        ][tenor]
    ) {

        return simulationInsuranceRate[
            insuranceType
        ][tenor];

    }

    return 0;

}


/* =====================================================
   CALCULATE INSURANCE COVERAGE
===================================================== */

function calculateInsuranceCoverage(
    otrNett,
    tenor
) {

    const factor =
        simulationDepreciation[
            tenor
        ] || 1;

    return (
        otrNett *
        factor
    );

}


/* =====================================================
   CALCULATE INSURANCE PREMIUM
===================================================== */

function calculateInsurancePremium(
    otrNett,
    tenor,
    insuranceType
) {

    const coverage =
        calculateInsuranceCoverage(
            otrNett,
            tenor
        );


    /* =========================================
       COMPREHENSIVE
    ========================================= */

    if (
        insuranceType ===
        "comprehensive"
    ) {

        const rate =
            getInsuranceRate(
                "comprehensive",
                tenor
            );

        return {

            coverage:
                coverage,

            rate:
                rate,

            premium:
                coverage * rate

        };

    }


    /* =========================================
       TLO
    ========================================= */

    if (
        insuranceType ===
        "tlo"
    ) {

        const rate =
            getInsuranceRate(
                "tlo",
                tenor
            );

        return {

            coverage:
                coverage,

            rate:
                rate,

            premium:
                coverage * rate

        };

    }


    /* =========================================
       COMBINE
    ========================================= */

    if (
        insuranceType ===
        "combine"
    ) {

        const comprehensiveRate =
            getInsuranceRate(
                "comprehensive",
                12
            );


        const tloRate =
            getInsuranceRate(
                "tlo",
                tenor
            );


        const firstYearCoverage =
            calculateInsuranceCoverage(
                otrNett,
                12
            );


        const remainingCoverage =
            calculateInsuranceCoverage(
                otrNett,
                tenor
            );


        const firstYearPremium =
            firstYearCoverage *
            comprehensiveRate;


        const remainingPremium =
            remainingCoverage *
            tloRate;


        return {

            coverage:
                firstYearCoverage,

            rate:
                comprehensiveRate,

            premium:
                firstYearPremium +
                remainingPremium

        };

    }


    return {

        coverage: 0,

        rate: 0,

        premium: 0

    };

}

function getLeadStatusClass(status) {

    if (status === "Contacted") {
        return "status-contacted";
    }

    if (status === "Simulation") {
        return "status-simulation";
    }

    if (
        status === "Approved" ||
        status === "Valid/DO" ||
        status === "UF"
    ) {
        return "status-approved";
    }

    if (status === "Lost Opportunity") {
        return "status-lost";
    }

    return "";

}


function formatLeadDate(dateString) {

    if (!dateString) {
        return "-";
    }

    const date =
        new Date(dateString);

    if (Number.isNaN(date.getTime())) {
        return "-";
    }

    return date.toLocaleDateString(
        "id-ID",
        {
            day: "2-digit",
            month: "short",
            year: "numeric"
        }
    );

}

        document.getElementById("lead-empty-state");

        document.getElementById("total-lead");


    if (!leadTableBody) {
        return;
    }


    /*
     * ==========================================
     * AMBIL CUSTOMER YANG SUDAH MASUK KE TAF
     * ==========================================
     */

    const storedLeads =
        localStorage.getItem("tafLeads");

    let leads = [];


    if (storedLeads) {

        try {

            leads = JSON.parse(storedLeads);

            if (!Array.isArray(leads)) {
                leads = [];
            }

        } catch (error) {

            console.error(
                "Data customer tidak dapat dibaca:",
                error
            );

            leads = [];

        }

    }


    /*
     * ==========================================
     * TOTAL LEAD
     * ==========================================
     *
     * Total Lead = jumlah customer yang sudah
     * masuk / menunjukkan ketertarikan melalui
     * TAF Digital Showcase.
     *
     * Tidak peduli statusnya apa.
     *
     * New Lead
     * Contacted
     * Qualified Prospect
     * Simulation
     * App-In
     * Approved
     * Valid/DO
     * UF
     * Lost Opportunity
     *
     * Semuanya tetap dihitung sebagai customer
     * yang pernah masuk sebagai lead.
     */

    if (totalLeadElement) {

        totalLeadElement.textContent =
            leads.length;

    }


    /*
     * ==========================================
     * KALAU BELUM ADA CUSTOMER
     * ==========================================
     */

    if (leads.length === 0) {

        leadTableBody.innerHTML = "";

        if (emptyState) {
            emptyState.style.display = "block";
        }

        return;

    }


    /*
     * ==========================================
     * CUSTOMER SUDAH ADA
     * ==========================================
     */

    if (emptyState) {
        emptyState.style.display = "none";
    }


    /*
     * ==========================================
     * TAMPILKAN CUSTOMER
     * ==========================================
     */

    leadTableBody.innerHTML = leads.map(
        function (lead) {

            const dp =
                lead.dp !== undefined
                    ? lead.dp
                    : "-";


            const tenor =
                lead.tenor !== undefined
                    ? lead.tenor
                    : "-";

                lead.status || "New Lead";


            return `

                <tr>

                    <td>

                        <div class="lead-customer">

                            <strong>
                                ${lead.customerName || "-"}
                            </strong>

                            <span>
                                ${lead.customerPhone || "-"}
                            </span>

                        </div>

                    </td>


                    <td>

                        <div class="lead-unit">

                            <strong>
                                ${lead.vehicleName || "-"}
                            </strong>

                            <span>
                                ${formatLeadCurrency(
                                    lead.vehiclePrice
                                )}
                            </span>

                        </div>

                    </td>

                    </td>


                    <td>

                        <div class="lead-simulation">

                            <strong>
                                DP ${dp}%
                            </strong>

                            <span>
                                ${tenor} Bulan
                            </span>

                        </div>

                    </td>


                    <td>

                        <span class="lead-status-badge">

                            ${status}

                        </span>

                    </td>


                    <td>

                        <a
                            href="lead-detail.html?id=${encodeURIComponent(

                            )}"

function getLeadById() {

    const params = new URLSearchParams(window.location.search);
    const leadId = params.get("id");

    if (!leadId) {
        return null;
    }

    const storedLeads = localStorage.getItem("tafLeads");

    if (!storedLeads) {
        return null;
    }

    try {

        const leads = JSON.parse(storedLeads);

        if (!Array.isArray(leads)) {
            return null;
        }

        return leads.find(function (lead) {
            return lead.id === leadId;
        });

    } catch (error) {

        return null;

    }
}


function formatLeadDetailDate(dateString) {

    if (!dateString) {
        return "-";
    }

    const date = new Date(dateString);

    if (isNaN(date.getTime())) {
        return "-";
    }

    return date.toLocaleDateString("id-ID", {
        day: "2-digit",
        month: "long",
        year: "numeric"
    });
}


function formatLeadCurrency(value) {

    const number = Number(value);

    if (isNaN(number)) {
        return "-";
    }

    return "Rp " + number.toLocaleString("id-ID");
}

function updateLeadStatus() {

    const lead = getLeadById();

    if (!lead) {
        alert("Data lead tidak ditemukan.");
        return;
    }

    const newStatus =
        document.getElementById("lead-status-select").value;


    const storedLeads =
        localStorage.getItem("tafLeads");

    if (!storedLeads) {
        return;
    }


    try {

        const leads = JSON.parse(storedLeads);

        const leadIndex = leads.findIndex(function (item) {
            return item.id === lead.id;
        });


        if (leadIndex === -1) {
            alert("Lead tidak ditemukan.");
            return;
        }


        leads[leadIndex].status = newStatus;

        localStorage.setItem(
            "tafLeads",
            JSON.stringify(leads)
        );


        updateLeadStatusBadge(newStatus);

        alert("Status lead berhasil diperbarui.");


    } catch (error) {

        alert("Terjadi kesalahan saat memperbarui status.");

    }

}


function saveLeadFollowup() {

    const lead = getLeadById();

    if (!lead) {
        alert("Data lead tidak ditemukan.");
        return;
    }


    const noteElement =
        document.getElementById("lead-followup-note");

    if (!noteElement) {
        alert("Kolom catatan follow-up tidak ditemukan.");
        return;
    }


    const note =
        noteElement.value.trim();


    if (!note) {
        alert("Silakan isi catatan follow-up terlebih dahulu.");
        return;
    }


    const storedLeads =
        localStorage.getItem("tafLeads");


    if (!storedLeads) {
        alert("Data lead tidak ditemukan.");
        return;
    }


    try {

        const leads =
            JSON.parse(storedLeads);


        const leadIndex =
            leads.findIndex(function (item) {

                return item.id === lead.id;

            });


        if (leadIndex === -1) {

            alert("Lead tidak ditemukan.");

            return;
        }


        /*
         * ==========================================
         * BUAT HISTORY FOLLOW-UP
         * ==========================================
         */

        if (
            !Array.isArray(
                leads[leadIndex].followupHistory
            )
        ) {

            leads[leadIndex].followupHistory = [];

        }


        const followupData = {

            date: new Date().toISOString(),

            note: note,

            nextAction: "Follow-up lanjutan"

        };


        leads[leadIndex].followupHistory.push(
            followupData
        );


        /*
         * ==========================================
         * UPDATE DATA TERAKHIR
         * ==========================================
         */

        leads[leadIndex].lastFollowup =
            followupData.date;


        leads[leadIndex].followupNote =
            note;


        leads[leadIndex].nextAction =
            followupData.nextAction;


        /*
         * ==========================================
         * SIMPAN KE LOCAL STORAGE
         * ==========================================
         */

        localStorage.setItem(
            "tafLeads",
            JSON.stringify(leads)
        );


        /*
         * ==========================================
         * UPDATE SUMMARY
         * ==========================================
         */

        const lastFollowupElement =
            document.getElementById(
                "lead-last-followup"
            );


        const nextActionElement =
            document.getElementById(
                "lead-next-action"
            );


        if (lastFollowupElement) {

            lastFollowupElement.textContent =
                formatLeadDetailDate(
                    followupData.date
                );

        }


        if (nextActionElement) {

            nextActionElement.textContent =
                followupData.nextAction;

        }


        /*
         * ==========================================
         * REFRESH RIWAYAT FOLLOW-UP
         * ==========================================
         */

        renderLeadFollowupHistory(
            leads[leadIndex]
        );


        /*
         * ==========================================
         * KOSONGKAN TEXTAREA
         * ==========================================
         */

        noteElement.value = "";


        alert(
            "Follow-up berhasil disimpan."
        );


    } catch (error) {

        console.error(
            "Error saat menyimpan follow-up:",
            error
        );


        alert(
            "Terjadi kesalahan saat menyimpan follow-up."
        );

    }

}

function renderLeadFollowupHistory(lead) {

    const historyContainer =
        document.getElementById(
            "lead-followup-history-list"
        );


    if (!historyContainer) {
        return;
    }


    /*
     * Ambil history follow-up
     */

    let history =
        Array.isArray(lead.followupHistory)
            ? lead.followupHistory
            : [];


    /*
     * ==========================================
     * MIGRASI DATA LAMA
     * ==========================================
     *
     * Kalau sebelumnya sudah pernah menyimpan
     * followupNote dengan sistem lama, tetapi
     * belum memiliki followupHistory, kita tetap
     * tampilkan catatan tersebut.
     */

    if (
        history.length === 0 &&
        lead.followupNote
    ) {

        history = [

            {
                date:
                    lead.lastFollowup ||
                    lead.createdAt,

                note:
                    lead.followupNote,

                nextAction:
                    lead.nextAction ||
                    "Follow-up lanjutan"

            }

        ];

    }


    /*
     * Kalau belum ada history
     */

    if (history.length === 0) {

        historyContainer.innerHTML = `

            <div class="lead-followup-empty">

                Belum ada riwayat follow-up.
                <br>

                Simpan hasil komunikasi pertama
                untuk mulai mencatat aktivitas customer.

            </div>

        `;

        return;

    }


    /*
     * Tampilkan history terbaru di atas
     */

    const sortedHistory =
        [...history].reverse();


    historyContainer.innerHTML =
        sortedHistory.map(
            function (item) {

                return `

                    <div class="lead-followup-history-item">

                        <div class="lead-followup-history-meta">

                            <span class="lead-followup-history-date">

                                ${formatLeadDetailDate(
                                    item.date
                                )}

                            </span>


                            <span class="lead-followup-history-action">

                                Next Action:
                                ${escapeLeadFollowupText(
                                    item.nextAction ||
                                    "Follow-up lanjutan"
                                )}

                            </span>

                        </div>


                        <p class="lead-followup-history-note">

                            ${escapeLeadFollowupText(
                                item.note
                            )}

                        </p>

                    </div>

                `;

            }
        ).join("");

}

function escapeLeadFollowupText(text) {

    if (text === null || text === undefined) {
        return "";
    }


    return String(text)

        .replace(/&/g, "&amp;")

        .replace(/</g, "&lt;")

        .replace(/>/g, "&gt;")

        .replace(/"/g, "&quot;")

        .replace(/'/g, "&#039;");

}
